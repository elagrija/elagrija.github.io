HMODULE LoadOriginalDll() {
    static HMODULE hOrig = NULL;
    if (!hOrig) {
        hOrig = LoadLibraryW(ORIGINAL_COM_DLL_PATH);
    }
    return hOrig;
}

extern "C" __declspec(dllexport) HRESULT WINAPI MyDllCanUnloadNow() {
    //Go();
    HMODULE hOrigDLL = LoadOriginalDll();
    if (!hOrigDLL) return S_FALSE;

    tDllCanUnloadNow pFunc = (tDllCanUnloadNow)GetProcAddress(hOrigDLL, "DllCanUnloadNow");
    //if (!pFunc) return S_FALSE;

    return pFunc();
}

extern "C" __declspec(dllexport) HRESULT WINAPI MyDllGetActivationFactory(HSTRING classId, IActivationFactory** factory) {
    //Go();
    HMODULE hOrigDLL = LoadOriginalDll();

    tDllGetActivationFactory pFunc = (tDllGetActivationFactory)GetProcAddress(hOrigDLL, "DllGetActivationFactory");
    //if (!pFunc) return S_FALSE;

    return pFunc(classId, factory);
}

extern "C" __declspec(dllexport) HRESULT WINAPI MyDllGetClassObject(REFCLSID rclsid, REFIID riid, LPVOID FAR* ppv) {
    Go();

}
